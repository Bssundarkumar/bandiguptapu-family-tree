        targetEl.classList.add('highlight');
        showPanel(el.dataset.jump, false);
      }
    });
  });

  panel.querySelectorAll('.edit-btn').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      showEditForm(btn.dataset.action, btn.dataset.id);
    });
  });
}

/* ---------------- EN -> TELUGU TRANSLITERATION ---------------- */
function transliterateToTelugu(input){
  if(!input) return input;
  if(/[^\x00-\x7F]/.test(input)) return input;

  const VOWELS_INDEP = [
    ['aa','ఆ'],['ee','ఈ'],['ii','ఈ'],['oo','ఊ'],['uu','ఊ'],
    ['ai','ఐ'],['au','ఔ'],
    ['a','అ'],['i','ఇ'],['u','ఉ'],['e','ఎ'],['o','ఒ']
  ];
  const VOWELS_MATRA = [
    ['aa','ా'],['ee','ీ'],['ii','ీ'],['oo','ూ'],['uu','ూ'],
    ['ai','ై'],['au','ౌ'],
    ['a',''],['i','ి'],['u','ు'],['e','ె'],['o','ొ']
  ];
  // Case-SENSITIVE, longest-first. Uppercase = retroflex/special; lowercase = common default.
  const CONSONANTS = [
    ['ksh','క్ష'],['gy','జ్ఞ'],['jn','జ్ఞ'],
    ['kh','ఖ'],['gh','ఘ'],['chh','ఛ'],['ch','చ'],['jh','ఝ'],['ny','ఞ'],
    ['th','థ'],['dh','ధ'],['ph','ఫ'],['bh','భ'],['shh','ష'],['sh','శ'],['Sh','ష'],
    ['T','ట'],['Th','ఠ'],['D','డ'],['Dh','ఢ'],['N','ణ'],['L','ళ'],
    ['k','క'],['g','గ'],['c','చ'],['j','జ'],
    ['t','త'],['d','ద'],['n','న'],
    ['p','ప'],['f','ఫ'],['b','బ'],['m','మ'],
    ['y','య'],['r','ర'],['l','ల'],['v','వ'],['w','వ'],
    ['s','స'],['h','హ'],['z','జ'],['x','క్ష'],['q','క']
  ].sort((a,b)=>b[0].length-a[0].length);
  const VIRAMA = '్', ANUSVARA = 'ం';

  const SPECIAL_CASE_KEYS = new Set(['T','Th','D','Dh','N','L','Sh']); // these MUST stay uppercase to mean retroflex; everything else matches case-insensitively
  function matchCS(pos, list){
    const atWordStart = pos===0 || /[^a-zA-Z]/.test(input[pos-1]||' ');
    for(const [k,v] of list){
      if(SPECIAL_CASE_KEYS.has(k)){
        if(atWordStart) continue; // never treat a word-initial capital as an intentional retroflex marker
        if(input.substr(pos,k.length) === k) return [k, v];
      } else {
        if(input.substr(pos,k.length).toLowerCase() === k.toLowerCase()) return [k, v];
      }
    }
    return null;
  }
  function isConsLetter(pos){
    // does a consonant token start at pos (case-sensitive)?
    return !!matchCS(pos, CONSONANTS);
  }
  function isVowelChar(c){ return /[aeiouAEIOU]/.test(c||''); }

  let out = '';
  let i = 0;
  const s = input;
  const n = s.length;

  while(i < n){
    const ch = s[i];
    if(/[^a-zA-Z]/.test(ch)){ out += ch; i++; continue; }

    // Gemination: identical consecutive letters (yy, kk, tt, ll, ss, pp, bb, gg, vv, rr, mm, nn), lowercase only
    const c0 = s[i], c1 = s[i+1];
    if(c1 && c0 === c1 && /[a-z]/.test(c0) && !isVowelChar(c0)){
      const single = matchCS(i, CONSONANTS) || (c0==='n'?['n','న']:c0==='m'?['m','మ']:null);
      if(single){
        const vow = matchCS(i+2, VOWELS_MATRA);
        out += single[1] + VIRAMA + single[1] + (vow ? vow[1] : '');
        i += 2 + (vow ? vow[0].length : 0);
        continue;
      }
    }

    const cons = matchCS(i, CONSONANTS);
    if(cons){
      let consChar = cons[1];
      i += cons[0].length;
      const vow = matchCS(i, VOWELS_MATRA);
      const hasVowel = !!vow;
      // n/m with inherent 'a' (no explicit vowel) and followed by another consonant -> anusvara
      if((cons[0]==='n' || cons[0]==='m') && !hasVowel && i<n && !isVowelChar(s[i]) && isConsLetter(i)){
        out += ANUSVARA;
        continue;
      }
      if(vow){ out += consChar + vow[1]; i += vow[0].length; }
      else { out += consChar; }
      continue;
    }
    const vind = matchCS(i, VOWELS_INDEP);
    if(vind){ out += vind[1]; i += vind[0].length; continue; }
    out += ch; i++;
  }
  return out;
}

function showEditForm(action, nodeId){
  const slot = document.getElementById('edit-form-slot');
  if(!slot) return;
  const labels = {
    son: {te:'కొడుకు జోడించు', en:'Add son'},
    spouse: {te:'జీవిత భాగస్వామి జోడించు', en:'Add spouse'},
    sibling: {te:'తోబుట్టువు జోడించు', en:'Add sibling'}
  };
  const l = labels[action];
  slot.innerHTML = `
    <div class="edit-form">
      <div class="ef-title telugu">${l.te} <span>(${l.en})</span></div>
      <div class="ef-field-row"><input type="text" id="ef-name" class="telugu" placeholder="పేరు · Name (required)"><button type="button" class="ef-translit-btn" data-target="ef-name" title="Convert English to Telugu">🔤</button></div>
      <div class="ef-field-row"><input type="text" id="ef-title" class="telugu" placeholder="బిరుదు/వృత్తి · Title (optional)"><button type="button" class="ef-translit-btn" data-target="ef-title" title="Convert English to Telugu">🔤</button></div>
      <input type="text" id="ef-dates" placeholder="తేదీలు · Dates (optional)">
      ${action!=='spouse' ? `<div class="ef-field-row"><input type="text" id="ef-spousename" class="telugu" placeholder="జీవిత భాగస్వామి పేరు · Their spouse's name (optional)"><button type="button" class="ef-translit-btn" data-target="ef-spousename" title="Convert English to Telugu">🔤</button></div>` : ''}
      <div class="ef-hint">Type in English, then tap 🔤 to convert to Telugu · ఇంగ్లీష్‌లో టైప్ చేసి 🔤 నొక్కండి</div>
      <div class="ef-btn-row">
        <button id="ef-save" class="ef-save">సేవ్ చేయి · Save</button>
        <button id="ef-cancel" class="ef-cancel">రద్దు · Cancel</button>
      </div>
    </div>`;
  document.getElementById('ef-name').focus();
  document.getElementById('ef-cancel').addEventListener('click', ()=>{ slot.innerHTML=''; });
  slot.querySelectorAll('.ef-translit-btn').forEach(btn=>{
    btn.addEventListener('click', ()=>{
      const inp = document.getElementById(btn.dataset.target);
      if(inp && inp.value.trim()) inp.value = transliterateToTelugu(inp.value);
      inp.focus();
    });
  });
  document.getElementById('ef-save').addEventListener('click', ()=>{
    const name = document.getElementById('ef-name').value.trim();
    if(!name){ document.getElementById('ef-name').focus(); return; }
    const title = document.getElementById('ef-title').value.trim();
    const dates = document.getElementById('ef-dates').value.trim();
    const spouseNameEl = document.getElementById('ef-spousename');
    const spouseName = spouseNameEl ? spouseNameEl.value.trim() : '';

    const opts = {};
    if(title) opts.title = title;
    if(dates) opts.dates = dates;
    if(spouseName) opts.spouse = {name: spouseName};

    const entry = index[nodeId];
    if(!entry) return;

    if(action === 'spouse'){
      entry.node.spouse = {name, title: title||''};
    } else if(action === 'son'){
      if(!entry.node.children) entry.node.children = [];
      entry.node.children.push(n(name, opts));
    } else if(action === 'sibling'){
      if(entry.parent){
        if(!entry.parent.children) entry.parent.children = [];
        entry.parent.children.push(n(name, opts));
      }
    }
    renderAll(true);
    showPanel(nodeId, false);
  });
}

/* ---------------- ZOOM & PAN ---------------- */
let zoom = 1;
function applyZoom(){ canvas.style.transform = `scale(${zoom})`; }
function resetZoom(){ zoom = 1; applyZoom(); }
document.getElementById('zoom-in').addEventListener('click', ()=>{ zoom = Math.min(zoom+0.1, 1.6); applyZoom(); });
document.getElementById('zoom-out').addEventListener('click', ()=>{ zoom = Math.max(zoom-0.1, 0.4); applyZoom(); });
