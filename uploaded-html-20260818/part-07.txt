
const wrap = document.getElementById('canvas-wrap');
wrap.addEventListener('wheel', (e)=>{
  if(!e.ctrlKey) return;
  e.preventDefault();
  zoom = Math.min(1.6, Math.max(0.4, zoom + (e.deltaY<0?0.08:-0.08)));
  applyZoom();
}, {passive:false});

let isDown=false, startX, startY, scrollLeft, scrollTop;
wrap.addEventListener('mousedown', (e)=>{
  isDown=true; wrap.classList.add('dragging');
  startX = e.pageX; startY = e.pageY;
  scrollLeft = wrap.scrollLeft; scrollTop = wrap.scrollTop;
});
window.addEventListener('mouseup', ()=>{ isDown=false; wrap.classList.remove('dragging'); });
wrap.addEventListener('mouseleave', ()=>{ isDown=false; wrap.classList.remove('dragging'); });
wrap.addEventListener('mousemove', (e)=>{
  if(!isDown) return;
  e.preventDefault();
  wrap.scrollLeft = scrollLeft - (e.pageX - startX);
  wrap.scrollTop = scrollTop - (e.pageY - startY);
});

/* Touch support: one-finger pan, two-finger pinch-to-zoom (for phones/tablets/iPads) */
let touchMode = null; // 'pan' | 'pinch'
let touchStartX=0, touchStartY=0, touchScrollLeft=0, touchScrollTop=0;
let pinchStartDist=0, pinchStartZoom=1;

function touchDist(t0,t1){
  const dx = t0.clientX - t1.clientX, dy = t0.clientY - t1.clientY;
  return Math.sqrt(dx*dx + dy*dy);
}

wrap.addEventListener('touchstart', (e)=>{
  if(e.touches.length === 1){
    touchMode = 'pan';
    touchStartX = e.touches[0].pageX;
    touchStartY = e.touches[0].pageY;
    touchScrollLeft = wrap.scrollLeft;
    touchScrollTop = wrap.scrollTop;
  } else if(e.touches.length === 2){
    touchMode = 'pinch';
    pinchStartDist = touchDist(e.touches[0], e.touches[1]);
    pinchStartZoom = zoom;
  }
}, {passive:true});

wrap.addEventListener('touchmove', (e)=>{
  if(touchMode === 'pan' && e.touches.length === 1){
    e.preventDefault();
    wrap.scrollLeft = touchScrollLeft - (e.touches[0].pageX - touchStartX);
    wrap.scrollTop = touchScrollTop - (e.touches[0].pageY - touchStartY);
  } else if(touchMode === 'pinch' && e.touches.length === 2){
    e.preventDefault();
    const newDist = touchDist(e.touches[0], e.touches[1]);
    const ratio = newDist / (pinchStartDist || newDist);
    zoom = Math.min(1.6, Math.max(0.4, pinchStartZoom * ratio));
    applyZoom();
  }
}, {passive:false});

wrap.addEventListener('touchend', (e)=>{
  if(e.touches.length === 0) touchMode = null;
  else if(e.touches.length === 1){
    // dropped from pinch to single-finger — restart pan reference point
    touchMode = 'pan';
    touchStartX = e.touches[0].pageX;
    touchStartY = e.touches[0].pageY;
    touchScrollLeft = wrap.scrollLeft;
    touchScrollTop = wrap.scrollTop;
  }
});

/* ---------------- EXPORT DATA (serialize live tree back to n(...) code) ---------------- */
function esc(s){ return String(s||'').replace(/\\/g,'\\\\').replace(/'/g,"\\'"); }
function serializeNode(node, indent){
  const pad = '  '.repeat(indent);
  const opts = [];
  if(node.title) opts.push(`title:'${esc(node.title)}'`);
  if(node.dates) opts.push(`dates:'${esc(node.dates)}'`);
  if(node.note) opts.push(`note:'${esc(node.note)}'`);
  if(node.placeholder) opts.push(`placeholder:true`);
  if(node.spouse) opts.push(`spouse:{name:'${esc(node.spouse.name)}'${node.spouse.title?`, title:'${esc(node.spouse.title)}'`:''}}`);
  if(node.children && node.children.length){
    const kids = node.children.map(c=>serializeNode(c, indent+1)).join(',\n');
    opts.push(`children:[\n${kids}\n${pad}]`);
  }
  const optsStr = opts.length ? `, {${opts.join(', ')}}` : '';
  return `${pad}n('${esc(node.name)}'${optsStr})`;
}
function exportAllData(){
  let out = '';
  out += '// ===== branch1 (పద్దప్ప) =====\n';
  out += `const branch1 = ${serializeNode(branch1,0).trim()};\n\n`;
  out += '// ===== branch2 (నరసప్ప) =====\n';
  out += `const branch2 = ${serializeNode(branch2,0).trim()};\n\n`;
  out += '// ===== branch3 (వెంకప్ప గ్రధ్రుప్యాలికి — position unconfirmed) =====\n';
  out += `const branch3 = ${serializeNode(branch3,0).trim()};\n\n`;
  out += '// ===== branch4 (వెంకప్ప గుర్రపుస్వారీకి / Son 3) =====\n';
  out += `const branch4 = ${serializeNode(branch4,0).trim()};\n`;
  return out;
}

const searchInput = document.getElementById('search');
const resultsEl = document.getElementById('search-results');
searchInput.addEventListener('input', ()=>{
  const q = searchInput.value.trim();
  if(!q){ resultsEl.classList.remove('show'); return; }
  const matches = Object.values(index).filter(e => e.node.name.includes(q)).slice(0,20);
  if(!matches.length){
    resultsEl.innerHTML = `<div class="sr-item telugu">ఫలితాలు లేవు · no matches</div>`;
  } else {
    resultsEl.innerHTML = matches.map(m=>`
      <div class="sr-item telugu" data-id="${m.node.id}" data-branch="${m.branchKey}">
        ${m.node.name}
        <div class="sr-branch">${branches[m.branchKey].label}</div>
      </div>`).join('');
  }
  resultsEl.classList.add('show');
  resultsEl.querySelectorAll('.sr-item[data-id]').forEach(el=>{
    el.addEventListener('click', ()=>{
      const branchKey = el.dataset.branch;
      const id = el.dataset.id;
      resultsEl.classList.remove('show');
      searchInput.value='';
      if(activeBranch !== branchKey){ activeBranch = branchKey; renderAll(); }
      requestAnimationFrame(()=>{
        const targetEl = canvas.querySelector(`.node[data-id="${id}"]`);
        if(targetEl){
          targetEl.scrollIntoView({behavior:'smooth', block:'center', inline:'center'});
          targetEl.classList.add('highlight');
          showPanel(id, false);
        }
      });
    });
  });
});
document.addEventListener('click', (e)=>{
  if(!e.target.closest('.search-wrap')) resultsEl.classList.remove('show');
});
