/* ---------------- INDEX (for search & parent lookups) ---------------- */
const index = {}; // id -> {node, parent, branchKey}
function buildIndex(node, parent, branchKey){
  index[node.id] = {node, parent, branchKey};
  (node.children||[]).forEach(c => buildIndex(c, node, branchKey));
}
Object.entries(branches).forEach(([key,b])=>{ if(b.root) buildIndex(b.root, null, key); });
function reindexActive(){
  for(const k in index) delete index[k];
  const b = branches[activeBranch];
  if(b.root) buildIndex(b.root, null, activeBranch);
}

/* ---------------- RENDER TABS ---------------- */
const tabsEl = document.getElementById('tabs');
let activeBranch = 'full';
Object.entries(branches).forEach(([key,b])=>{
  const t = document.createElement('div');
  t.className = 'tab telugu' + (key===activeBranch?' active':'') + (!b.enabled?' disabled':'');
  t.textContent = b.label;
  t.title = b.enabled ? '' : 'ఇంకా transcribe చేయలేదు — not yet transcribed';
  if(b.enabled){
    t.addEventListener('click', ()=>{ activeBranch = key; renderAll(); });
  }
  tabsEl.appendChild(t);
});

/* ---------------- RENDER TREE ---------------- */
const canvas = document.getElementById('canvas');

function nodeBoxHTML(node, isRoot){
  const classes = ['node'];
  if(isRoot) classes.push('root');
  if(node.placeholder) classes.push('placeholder');
  const titleHTML = node.title ? `<span class="n-title telugu">${node.title}</span>` : (node.dates ? `<span class="n-title">${node.dates}</span>` : '');
  return `<div class="${classes.join(' ')} telugu" data-id="${node.id}">${node.name}${titleHTML}</div>`;
}

function renderNode(node, isRoot=false){
  let slot = `<div class="node-slot">${nodeBoxHTML(node, isRoot)}`;
  if(node.spouse){
    slot += `<span class="spouse-link"></span><div class="node spouse telugu" data-id="${node.id}-spouse">${node.spouse.name}${node.spouse.title?`<span class="n-title telugu">${node.spouse.title}</span>`:''}</div>`;
  }
  slot += `</div>`;
  let childrenHTML = '';
  if(node.children && node.children.length){
    childrenHTML = `<ul>${node.children.map(c=>`<li>${renderNode(c)}</li>`).join('')}</ul>`;
  }
  return slot + childrenHTML;
}

function renderAll(preserveView){
  // tabs active state
  [...tabsEl.children].forEach((t,i)=>{
    const key = Object.keys(branches)[i];
    t.classList.toggle('active', key===activeBranch);
  });

  const b = branches[activeBranch];
  if(!b.root){
    canvas.innerHTML = `<div class="empty-branch telugu">ఈ శాఖ ఇంకా transcribe చేయలేదు.<br><br><span style="font-family:'Fraunces',serif; font-style:italic; color:var(--ink-soft); font-size:14px;">This branch hasn't been transcribed from the original chart yet — tell Claude which ancestor to zoom in on next and it'll be added here.</span></div>`;
    return;
  }
  reindexActive();
  canvas.innerHTML = `<ul class="tree"><li>${renderNode(b.root, true)}</li></ul>`;
  if(!preserveView) resetZoom();
  attachNodeHandlers();
}

/* ---------------- NODE CLICK -> DETAIL PANEL ---------------- */
function attachNodeHandlers(){
  canvas.querySelectorAll('.node').forEach(el=>{
    el.addEventListener('click', (e)=>{
      e.stopPropagation();
      canvas.querySelectorAll('.node.highlight').forEach(x=>x.classList.remove('highlight'));
      el.classList.add('highlight');
      const id = el.dataset.id.replace('-spouse','');
      showPanel(id, el.dataset.id.endsWith('-spouse'));
    });
  });
}
